  <canvas id="canvas"></canvas>

  <script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    const upload = document.getElementById('upload');

    function drawHexagon(x, y, size, color) {
      const angle = Math.PI / 3; // 60 degrees
      ctx.beginPath();
      for (let i = 0; i < 6; i++) {
        ctx.lineTo(x + size * Math.cos(angle * i), y + size * Math.sin(angle * i));
      }
      ctx.closePath();
      ctx.fillStyle = color;
      ctx.fill();
    }

    function hexPixelate(img, hexSize) {
      canvas.width = img.width;
      canvas.height = img.height;

      // Draw the image temporarily
      const tempCanvas = document.createElement('canvas');
      const tempCtx = tempCanvas.getContext('2d');
      tempCanvas.width = img.width;
      tempCanvas.height = img.height;
      tempCtx.drawImage(img, 0, 0);

      const imgData = tempCtx.getImageData(0, 0, img.width, img.height).data;

      const hexHeight = Math.sqrt(3) * hexSize;
      for (let y = 0; y < img.height; y += hexHeight) {
        for (let x = 0; x < img.width; x += hexSize * 1.5) {
          let offsetX = (Math.floor(y / hexHeight) % 2) * hexSize * 0.75;
          let px = Math.floor(x + offsetX);
          let py = Math.floor(y);

          if (px >= img.width || py >= img.height) continue;

          const idx = (py * img.width + px) * 4;
          const r = imgData[idx];
          const g = imgData[idx + 1];
          const b = imgData[idx + 2];
          const color = `rgb(${r},${g},${b})`;

          drawHexagon(px, py, hexSize, color);
        }
      }
    }

    upload.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) return;

      const img = new Image();
      img.onload = () => hexPixelate(img, 10); // hexSize = 10
      img.src = URL.createObjectURL(file);
    });
  </script>
